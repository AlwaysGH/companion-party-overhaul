//===== Ragnarok Offline: companion-party-overhaul, fallen companions return
//= A fallen companion stays a corpse in the party while its owner is on the
//= same map, for a Priest companion, Death Valley or a Yggdrasil Leaf. That
//= part is the game's rule and is left alone.
//=
//= Once the owner leaves the map, the engine (app 1.5.4) means to bring the
//= companion back beside them, but two bugs get in the way:
//=  1. It keeps it in the party and sets a 2 s respawn timer, but the check
//=     runs every 100 ms and deletes and re-adds that timer each time, so it
//=     never fires: the companion stays dead until a relog.
//=  2. If the check runs while the owner is between maps (warping, or going
//=     back to the save point after a wipe), the owner counts as gone, and
//=     the dead companion is released from the party instead ("dead
//=     companion ... left behind by its owner; releasing it").
//=
//= Every 2 s, for each player:
//=  1. a companion dead on a different map from its owner is revived with
//=     `recovery`; the engine then sees a live companion left behind and
//=     brings it to its owner ("followed its owner to another map");
//=  2. a companion that is on duty (active in the engine's saved list,
//=     cp_companion_persistence) but not in the party was released: it is
//=     called back with `@companion summon`, the engine's own recall. One you
//=     bench yourself is off duty and stays benched. Reading the saved list
//=     catches a release however fast it happens (1.3.1 only caught
//=     companions it had seen lying dead, and missed quick ones).
//===========================================================================
-	script	PopCompanionRevive	-1,{
	end;

OnInit:
	.mod$ = "companion-party-overhaul";
	.on = callfunc("F_ModSetting", .mod$, "companion_return", 1);
	.say = callfunc("F_ModSetting", .mod$, "return_announce", 1);
	initnpctimer;	// the tick below
	end;

// The tick: one NPC timer for everyone, every 2000 ms. `addrid 0, 1` runs the
// tick for each player but skips anyone busy in another script (an NPC
// dialog). Before, each player had a timer of their own, and during a long
// dialog those queued up until the queue was full and were dropped ("event
// queue is full"), which stopped them for good.
OnTimer2000:
	initnpctimer;
	if (!.on)
		end;
	addrid 0, 1;
	if (getcharid(3) >= 95000000)
		end;	// an AI player, not a person
	goto OnTick;

OnPCLoginEvent:
	@cr_login = gettimetick(2);	// the engine recalls companions itself at login
	@cr_missing$ = "";
	@cr_tries = 0;
	end;

OnTick:
	.@me = getcharid(3);
	getmapxy(.@mymap$, .@x, .@y, BL_PC);
	.@n = 0;
	if (getcharid(1)) {
		getpartymember getcharid(1), 0;
		getpartymember getcharid(1), 1;
		getpartymember getcharid(1), 2;
		.@n = $@partymembercount;
		copyarray .@name$[0], $@partymembername$[0], .@n;
		copyarray .@cid[0], $@partymembercid[0], .@n;
		copyarray .@aid[0], $@partymemberaid[0], .@n;
	}

	// 1. Dead on another map but still in the party: revive, and the engine
	// brings them over.
	for (.@i = 0; .@i < .@n; .@i++) {
		if (.@cid[.@i] < 95000000 || .@aid[.@i] == .@me)
			continue;
		if (!isloggedin(.@aid[.@i], .@cid[.@i]))
			continue;
		if (!isdead(.@aid[.@i]))
			continue;
		// Where does the corpse lie? (attachrid works on a shell)
		.@where$ = "";
		if (attachrid(.@aid[.@i], 1))
			getmapxy(.@where$, .@x, .@y, BL_PC);
		attachrid(.@me);
		if (.@where$ == "" || .@where$ == .@mymap$)
			continue;	// same map: it waits for a Priest or a Leaf
		recovery 0, .@cid[.@i], 4;	// 4: revive the dead only
		if (.say && !isdead(.@aid[.@i]))
			dispbottom "[Companions] " + .@name$[.@i] + " is back on their feet and catching up with you.", 0x99DDFF;
	}

	// 2. On duty but not in the party: released by the engine. Not in the
	// first 15 s after login (the engine is recalling them), and not while
	// you're dead (it would only lose them again).
	if (gettimetick(2) - @cr_login < 15 || Hp < 1)
		end;
	.@k = query_sql("SELECT `name` FROM `cp_companion_persistence` WHERE `owner_account_id` = " + getcharid(3)
		+ " AND `owner_char_id` = " + getcharid(0) + " AND `active` = 1", .@duty$);
	.@missing$ = "";
	for (.@j = 0; .@j < .@k; .@j++) {
		.@here = 0;
		for (.@i = 0; .@i < .@n; .@i++) {
			if (.@cid[.@i] >= 95000000 && .@name$[.@i] == .@duty$[.@j]) {
				.@here = 1;
				break;
			}
		}
		if (!.@here)
			.@missing$ += (.@missing$ != "" ? ", " : "") + .@duty$[.@j];
	}
	if (.@missing$ == "") {
		@cr_missing$ = "";
		@cr_tries = 0;
		end;
	}
	// The same ones missing as last time: count the tries, and give up after
	// three until something changes (a map where they can't come, say).
	if (.@missing$ != @cr_missing$) {
		@cr_missing$ = .@missing$;
		@cr_tries = 0;
		end;	// wait one tick: the engine may still be finishing the release
	}
	if (@cr_tries >= 3)
		end;
	@cr_tries++;
	// One summon recalls every companion on duty that isn't with you.
	.@first$ = .@missing$;
	if (compare(.@first$, ", "))
		.@first$ = substr(.@first$, 0, strpos(.@first$, ",") - 1);
	atcommand "@companion summon " + .@first$;
	if (.say && @cr_tries == 1)
		dispbottom "[Companions] " + .@missing$ + (compare(.@missing$, ", ") ? " are" : " is") + " back on their feet and catching up with you.", 0x99DDFF;
	end;
}
